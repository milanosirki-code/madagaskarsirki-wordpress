if ( ! function_exists( 'mdg_ai_programs_list' ) ) {
    function mdg_ai_programs_list( $input = array() ) {
        $input = is_array( $input ) ? $input : array();
        $rows = MMC_Program_Service::all_programs();
        $items = array();

        $status   = isset( $input['status'] ) ? sanitize_key( $input['status'] ) : '';
        $province = isset( $input['province_name'] ) ? sanitize_text_field( $input['province_name'] ) : '';
        $district = isset( $input['district_name'] ) ? sanitize_text_field( $input['district_name'] ) : '';
        $search   = isset( $input['search'] ) ? sanitize_text_field( $input['search'] ) : '';
        $limit    = isset( $input['limit'] ) ? max( 1, min( 100, absint( $input['limit'] ) ) ) : 50;

        foreach ( (array) $rows as $row ) {
            if ( $status && $row->status !== $status ) {
                continue;
            }
            if ( $province && 0 !== strcasecmp( remove_accents( $row->province_name ), remove_accents( $province ) ) ) {
                continue;
            }
            if ( $district && 0 !== strcasecmp( remove_accents( $row->district_name ), remove_accents( $district ) ) ) {
                continue;
            }
            if ( $search ) {
                $haystack = $row->program_code . ' ' . $row->province_name . ' ' . $row->district_name . ' ' . (string) $row->notes;
                if ( false === mb_stripos( $haystack, $search, 0, 'UTF-8' ) ) {
                    continue;
                }
            }
            $items[] = mdg_ai_program_to_array( $row );
            if ( count( $items ) >= $limit ) {
                break;
            }
        }

        return array(
            'count'    => count( $items ),
            'items'    => $items,
            'statuses' => MMC_Program_Service::statuses(),
        );
    }
}

if ( ! function_exists( 'mdg_ai_program_get' ) ) {
    function mdg_ai_program_get( $input ) {
        $program_id = absint( $input['program_id'] ?? 0 );
        $program = MMC_Program_Service::get_program( $program_id );
        if ( ! $program ) {
            return new WP_Error( 'mdg_ai_program_missing', 'Program bulunamadı.' );
        }

        $venues = array();
        foreach ( (array) MMC_Venue_Service::venues_for_program( $program_id ) as $venue ) {
            $venues[] = mdg_ai_venue_to_array( $venue );
        }

        $event = array();
        if ( class_exists( 'MMC_Event_Service' ) && method_exists( 'MMC_Event_Service', 'event_for_program' ) ) {
            $event_row = MMC_Event_Service::event_for_program( $program_id );
            if ( $event_row ) {
                $event = array(
                    'id'               => (int) $event_row->id,
                    'event_code'       => (string) $event_row->event_code,
                    'event_title'      => (string) $event_row->event_title,
                    'event_date'       => (string) ( $event_row->event_date ?: '' ),
                    'status'           => (string) $event_row->status,
                    'program_venue_id' => isset( $event_row->program_venue_id ) ? (int) $event_row->program_venue_id : 0,
                    'mdg_event_id'     => isset( $event_row->mdg_event_id ) ? (int) $event_row->mdg_event_id : 0,
                );
            }
        }

        return array(
            'program' => mdg_ai_program_to_array( $program ),
            'venues'  => $venues,
            'event'   => $event,
        );
    }
}

if ( ! function_exists( 'mdg_ai_program_create' ) ) {
    function mdg_ai_program_create( $input ) {
        $data = array(
            'province_name' => sanitize_text_field( $input['province_name'] ?? '' ),
            'district_name' => sanitize_text_field( $input['district_name'] ?? '' ),
            'plan_year'     => absint( $input['plan_year'] ?? wp_date( 'Y' ) ),
            'planned_date'  => sanitize_text_field( $input['planned_date'] ?? '' ),
            'notes'         => sanitize_textarea_field( $input['notes'] ?? '' ),
        );

        if ( $data['planned_date'] && ! preg_match( '/^\d{4}-\d{2}-\d{2}$/', $data['planned_date'] ) ) {
            return new WP_Error( 'mdg_ai_invalid_date', 'planned_date YYYY-AA-GG biçiminde olmalıdır.' );
        }

        $program_id = MMC_Program_Service::create_program( $data );
        if ( is_wp_error( $program_id ) ) {
            return $program_id;
        }

        $program = MMC_Program_Service::get_program( $program_id );
        return array(
            'created' => true,
            'program' => mdg_ai_program_to_array( $program ),
        );
    }
}

if ( ! function_exists( 'mdg_ai_venue_candidates' ) ) {
    function mdg_ai_venue_candidates( $input ) {
        $program_id = absint( $input['program_id'] ?? 0 );
        if ( ! MMC_Program_Service::get_program( $program_id ) ) {
            return new WP_Error( 'mdg_ai_program_missing', 'Program bulunamadı.' );
        }

        $search = sanitize_text_field( $input['search'] ?? '' );
        $limit  = isset( $input['limit'] ) ? max( 1, min( 100, absint( $input['limit'] ) ) ) : 30;
        $items  = array();

        foreach ( (array) MMC_Venue_Service::venue_candidates_for_program( $program_id ) as $venue ) {
            if ( $search ) {
                $haystack = (string) $venue->venue_name . ' ' . (string) $venue->district_name . ' ' . (string) $venue->address;
                if ( false === mb_stripos( $haystack, $search, 0, 'UTF-8' ) ) {
                    continue;
                }
            }
            $item = mdg_ai_venue_to_array( $venue );
            $item['exact_district'] = ! empty( $venue->mmc_exact_district );
            $items[] = $item;
            if ( count( $items ) >= $limit ) {
                break;
            }
        }

        return array(
            'program_id' => $program_id,
            'count'      => count( $items ),
            'items'      => $items,
        );
    }
}

if ( ! function_exists( 'mdg_ai_private_static_call' ) ) {
    function mdg_ai_private_static_call( $class, $method, $args = array() ) {
        try {
            $ref = new ReflectionMethod( $class, $method );
            if ( ! $ref->isPublic() ) {
                $ref->setAccessible( true );
            }
            return $ref->invokeArgs( null, $args );
        } catch ( Throwable $e ) {
            return new WP_Error(
                'mdg_ai_private_call_failed',
                'Salon servis yordamcısı çalıştırılamadı: ' . $method
            );
        }
    }
}

if ( ! function_exists( 'mdg_ai_place_key' ) ) {
    function mdg_ai_place_key( $value ) {
        $value = trim( (string) $value );
        $value = remove_accents( $value );
        if ( function_exists( 'mb_strtolower' ) ) {
            $value = mb_strtolower( $value, 'UTF-8' );
        } else {
            $value = strtolower( $value );
        }
        return preg_replace( '/[^a-z0-9]+/u', '', $value );
    }
}

if ( ! function_exists( 'mdg_ai_resolve_province_code' ) ) {
    function mdg_ai_resolve_province_code( $province_code = '', $province_name = '' ) {
        $provinces = (array) MDG_Venues::provinces();

        $province_code = sanitize_text_field( (string) $province_code );
        if ( $province_code && isset( $provinces[ $province_code ] ) ) {
            return $province_code;
        }

        $needle = mdg_ai_place_key( $province_name );
        if ( $needle ) {
            foreach ( $provinces as $code => $name ) {
                if ( mdg_ai_place_key( $name ) === $needle ) {
                    return (string) $code;
                }
            }
        }

        return '';
    }
}

if ( ! function_exists( 'mdg_ai_venues_list' ) ) {
    function mdg_ai_venues_list( $input = array() ) {
        $input = is_array( $input ) ? $input : array();
        $active_only = array_key_exists( 'active_only', $input ) ? (bool) $input['active_only'] : true;
        $province = sanitize_text_field( $input['province_name'] ?? '' );
        $district = sanitize_text_field( $input['district'] ?? '' );
        $search   = sanitize_text_field( $input['search'] ?? '' );
        $limit    = isset( $input['limit'] ) ? max( 1, min( 200, absint( $input['limit'] ) ) ) : 50;

        $items = array();
        foreach ( (array) MDG_Venues::all( $active_only ) as $venue ) {
            if ( $province && mdg_ai_place_key( $venue->province_name ?? '' ) !== mdg_ai_place_key( $province ) ) {
                continue;
            }
            if ( $district && mdg_ai_place_key( $venue->district ?? '' ) !== mdg_ai_place_key( $district ) ) {
                continue;
            }
            if ( $search ) {
                $haystack = implode( ' ', array(
                    (string) ( $venue->name ?? '' ),
                    (string) ( $venue->province_name ?? '' ),
                    (string) ( $venue->district ?? '' ),
                    (string) ( $venue->address ?? '' ),
                    (string) ( $venue->contact_name ?? '' ),
                    (string) ( $venue->contact_phone ?? '' ),
                ) );
                if ( false === mb_stripos( $haystack, $search, 0, 'UTF-8' ) ) {
                    continue;
                }
            }
            $items[] = mdg_ai_master_venue_to_array( $venue );
            if ( count( $items ) >= $limit ) {
                break;
            }
        }

        return array(
            'count' => count( $items ),
            'items' => $items,
        );
    }
}

if ( ! function_exists( 'mdg_ai_venue_save' ) ) {
    function mdg_ai_venue_save( $input, $venue_id = 0 ) {
        global $wpdb;

        $venue_id = absint( $venue_id );
        $old = $venue_id ? MDG_Venues::get( $venue_id ) : null;

        if ( $venue_id && ! $old ) {
            return new WP_Error( 'mdg_ai_venue_missing', 'Güncellenecek salon bulunamadı.' );
        }

        $province_code = mdg_ai_resolve_province_code(
            $input['province_code'] ?? ( $old->province_code ?? '' ),
            $input['province_name'] ?? ( $old->province_name ?? '' )
        );
        $provinces = (array) MDG_Venues::provinces();
        $province_name = $province_code && isset( $provinces[ $province_code ] ) ? (string) $provinces[ $province_code ] : '';

        $district = sanitize_text_field(
            $input['district'] ?? ( $old->district ?? '' )
        );
        $district = MDG_Venues::canonical_district_name( $province_code, $district );

        $name = sanitize_text_field(
            $input['name'] ?? ( $old->name ?? '' )
        );

        $address = sanitize_textarea_field(
            $input['address'] ?? ( $old->address ?? '' )
        );

        $maps_raw = array_key_exists( 'maps_url', $input )
            ? (string) $input['maps_url']
            : (string) ( $old->maps_url ?? '' );

        $maps_url = mdg_ai_private_static_call(
            'MDG_Venues',
            'validate_maps_url',
            array( $maps_raw )
        );
        if ( is_wp_error( $maps_url ) ) {
            return $maps_url;
        }

        $latitude = array_key_exists( 'latitude', $input )
            ? ( null === $input['latitude'] || '' === $input['latitude'] ? null : (float) $input['latitude'] )
            : ( isset( $old->latitude ) ? ( null === $old->latitude ? null : (float) $old->latitude ) : null );

        $longitude = array_key_exists( 'longitude', $input )
            ? ( null === $input['longitude'] || '' === $input['longitude'] ? null : (float) $input['longitude'] )
            : ( isset( $old->longitude ) ? ( null === $old->longitude ? null : (float) $old->longitude ) : null );

        $data = array(
            'name'             => $name,
            'province_code'    => $province_code,
            'province_name'    => $province_name,
            'district'         => $district,
            'address'          => $address,
            'latitude'         => $latitude,
            'longitude'        => $longitude,
            'maps_url'         => $maps_url,
            'contact_name'     => sanitize_text_field( $input['contact_name'] ?? ( $old->contact_name ?? '' ) ),
            'contact_phone'    => sanitize_text_field( $input['contact_phone'] ?? ( $old->contact_phone ?? '' ) ),
            'default_capacity' => max( 1, absint( $input['default_capacity'] ?? ( $old->default_capacity ?? 500 ) ) ),
            'default_duration' => min( 360, max( 15, absint( $input['default_duration'] ?? ( $old->default_duration ?? 60 ) ) ) ),
            'notes'            => sanitize_textarea_field( $input['notes'] ?? ( $old->notes ?? '' ) ),
            'is_active'        => array_key_exists( 'is_active', $input ) ? ( $input['is_active'] ? 1 : 0 ) : ( $old ? (int) $old->is_active : 1 ),
            'updated_at'       => MDG_DB::now(),
        );

        if ( ! $data['name'] || ! $data['province_code'] || ! $data['province_name'] || ! $data['district'] || ! $data['address'] ) {
            return new WP_Error( 'mdg_ai_venue_required', 'İl, ilçe, salon adı ve adres zorunludur.' );
        }

        if ( ! MDG_Venues::district_is_valid( $province_code, $district ) ) {
            return new WP_Error( 'mdg_ai_venue_district', 'Seçilen ilçe bu ile ait görünmüyor.' );
        }

        $duplicate = mdg_ai_private_static_call(
            'MDG_Venues',
            'duplicate_id',
            array( $province_code, $district, $name, $venue_id )
        );
        if ( is_wp_error( $duplicate ) ) {
            return $duplicate;
        }
        if ( $duplicate ) {
            return new WP_Error( 'mdg_ai_venue_duplicate', 'Bu il / ilçe içinde aynı isimde bir salon zaten kayıtlı.' );
        }

        if ( null !== $latitude && ( $latitude < -90 || $latitude > 90 ) ) {
            return new WP_Error( 'mdg_ai_venue_latitude', 'Enlem -90 ile 90 arasında olmalıdır.' );
        }

        if ( null !== $longitude && ( $longitude < -180 || $longitude > 180 ) ) {
            return new WP_Error( 'mdg_ai_venue_longitude', 'Boylam -180 ile 180 arasında olmalıdır.' );
        }

        $table = MDG_DB::table( 'venues' );

        if ( $venue_id ) {
            $ok = $wpdb->update( $table, $data, array( 'id' => $venue_id ) );
            if ( false === $ok ) {
                return new WP_Error( 'mdg_ai_venue_update_failed', 'Salon güncellenemedi.' );
            }
            $action_key = 'venue.updated';
        } else {
            $data['created_at'] = MDG_DB::now();
            $ok = $wpdb->insert( $table, $data );
            if ( false === $ok ) {
                return new WP_Error( 'mdg_ai_venue_insert_failed', 'Salon kaydedilemedi.' );
            }
            $venue_id = (int) $wpdb->insert_id;
            $action_key = 'venue.created';
        }

        $audit = mdg_ai_private_static_call(
            'MDG_Venues',
            'audit',
            array( $action_key, $venue_id, $data )
        );

        $warning = mdg_ai_private_static_call(
            'MDG_Venues',
            'sync_location_qr_after_save',
            array( $venue_id, $old, $maps_url )
        );

        $saved = MDG_Venues::get( $venue_id );
        do_action( 'mdg_venue_saved', $venue_id, $saved );

        return array(
            'saved'   => true,
            'created' => ! $old,
            'warning' => is_wp_error( $warning ) ? $warning->get_error_message() : (string) $warning,
            'venue'   => mdg_ai_master_venue_to_array( $saved ),
        );
    }
}

if ( ! function_exists( 'mdg_ai_events_list' ) ) {
    function mdg_ai_events_list( $input = array() ) {
        $input = is_array( $input ) ? $input : array();
        $status   = sanitize_key( $input['status'] ?? '' );
        $province = sanitize_text_field( $input['province_name'] ?? '' );
        $district = sanitize_text_field( $input['district_name'] ?? '' );
        $search   = sanitize_text_field( $input['search'] ?? '' );
        $limit    = isset( $input['limit'] ) ? max( 1, min( 100, absint( $input['limit'] ) ) ) : 50;

        $items = array();

        foreach ( (array) MMC_Program_Service::all_programs() as $program ) {
            if ( $province && mdg_ai_place_key( $program->province_name ) !== mdg_ai_place_key( $province ) ) {
                continue;
            }
            if ( $district && mdg_ai_place_key( $program->district_name ) !== mdg_ai_place_key( $district ) ) {
                continue;
            }

            $event = MMC_Event_Service::event_for_program( $program->id );
            if ( ! $event ) {
                continue;
            }
            if ( $status && $event->status !== $status ) {
                continue;
            }
            if ( $search ) {
                $haystack = implode( ' ', array(
                    (string) $event->event_code,
                    (string) $event->event_title,
                    (string) $program->program_code,
                    (string) $program->province_name,
                    (string) $program->district_name,
                ) );
                if ( false === mb_stripos( $haystack, $search, 0, 'UTF-8' ) ) {
                    continue;
                }
            }

            $item = mdg_ai_event_to_array( $event );
            $item['program'] = mdg_ai_program_to_array( $program );
            $item['session_count'] = count( MMC_Event_Service::sessions( $event->id ) );
            $readiness = MMC_Event_Service::sales_readiness( $event->id );
            $item['sales_ready'] = ! empty( $readiness['ready'] );
            $item['sales_issues'] = isset( $readiness['issues'] ) ? array_values( $readiness['issues'] ) : array();
            $items[] = $item;

            if ( count( $items ) >= $limit ) {
                break;
            }
        }

        return array(
            'count'    => count( $items ),
            'items'    => $items,
            'statuses' => MMC_Event_Service::event_statuses(),
        );
    }
}

if ( ! function_exists( 'mdg_ai_event_get' ) ) {
    function mdg_ai_event_get( $input ) {
        $event_id = absint( $input['event_id'] ?? 0 );

            return new WP_Error( 'mdg_ai_event_missing', 'Etkinlik bulunamadı.' );

        $program = MMC_Program_Service::get_program( $event->program_id );
        $sessions = array();
        foreach ( (array) MMC_Event_Service::sessions( $event_id ) as $session ) {
            $sessions[] = mdg_ai_session_to_array( $session );

        $venue = array();
        if ( ! empty( $event->program_venue_id ) && class_exists( 'MMC_Venue_Service' ) ) {
            $venue_row = MMC_Venue_Service::get_program_venue( $event->program_venue_id );
            if ( $venue_row ) {
                $venue = mdg_ai_venue_to_array( $venue_row );
            }
        }

        return array(
            'event'           => mdg_ai_event_to_array( $event ),
            'program'         => $program ? mdg_ai_program_to_array( $program ) : array(),
            'venue'           => $venue,
            'sessions'        => $sessions,
            'sales_readiness' => MMC_Event_Service::sales_readiness( $event_id ),
            'integrations'     => array_map(
                function( $row ) {
                    return array(
                        'id'             => (int) $row->id,
                        'channel'        => (string) $row->channel,
                        'status'         => (string) $row->status,
                        'external_id'    => isset( $row->external_id ) ? (string) $row->external_id : '',
                        'external_url'   => isset( $row->external_url ) ? (string) $row->external_url : '',
                        'notes'          => isset( $row->notes ) ? (string) ( $row->notes ?: '' ) : '',
                        'last_synced_at' => isset( $row->last_synced_at ) ? (string) ( $row->last_synced_at ?: '' ) : '',

                (array) MMC_Event_Service::integrations( $event_id )

if ( ! function_exists( 'mdg_ai_ticket_type_update' ) ) {
    function mdg_ai_ticket_type_update( $input ) {
        $ticket_id = absint( $input['ticket_id'] ?? 0 );
        if ( ! $ticket_id ) {
            return new WP_Error( 'mdg_ai_ticket_id_required', 'ticket_id zorunludur.' );
        }

        $found = null;
        $event_id = 0;
        foreach ( (array) MMC_Program_Service::all_programs() as $program ) {
            $event = MMC_Event_Service::event_for_program( $program->id );
            if ( ! $event ) continue;
            foreach ( (array) MMC_Event_Service::ticket_types( $event->id ) as $ticket ) {
                if ( (int) $ticket->id === $ticket_id ) {
                    $found = $ticket;
                    $event_id = (int) $event->id;
                    break 2;
                }
            }
        }
        if ( ! $found ) {
            return new WP_Error( 'mdg_ai_ticket_missing', 'Bilet türü bulunamadı.' );
        }

        if ( 'family_2_2' === $found->ticket_code && array_key_exists( 'capacity_units', $input ) && 4 !== (int) $input['capacity_units'] ) {
            return new WP_Error(
                'mdg_ai_family_rule_locked',
                'Mevcut MMC servisi family_2_2 biletini 4 kapasite birimine sabitliyor. Aile paketi iş kuralı düzeltilmeden kapasite birimi değiştirilemez.'
            );
        }

        $data = array(
            'ticket_name'    => array_key_exists( 'ticket_name', $input ) ? sanitize_text_field( $input['ticket_name'] ) : $found->ticket_name,
            'price'          => array_key_exists( 'price', $input ) ? $input['price'] : $found->price,
            'capacity_units' => array_key_exists( 'capacity_units', $input ) ? absint( $input['capacity_units'] ) : (int) $found->capacity_units,
            'is_active'      => array_key_exists( 'is_active', $input ) ? (bool) $input['is_active'] : ! empty( $found->is_active ),
        );

        $ok = MMC_Event_Service::update_ticket_type( $ticket_id, $data );
        if ( is_wp_error( $ok ) ) return $ok;

        $updated = null;
        foreach ( (array) MMC_Event_Service::ticket_types( $event_id ) as $ticket ) {
            if ( (int) $ticket->id === $ticket_id ) {
                $updated = $ticket;
                break;
            }
        }
        return array(
            'updated' => true,
            'ticket'  => mdg_ai_ticket_to_array( $updated ),
        );
    }
}
