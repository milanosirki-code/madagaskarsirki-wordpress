/**
 * Var olan hukuki sayfayı birden fazla olası URL içinden bulur.
 */
if ( ! function_exists( 'ms_footer_find_page_url' ) ) {
    function ms_footer_find_page_url( $possible_slugs ) {
        foreach ( $possible_slugs as $slug ) {
            $page = get_page_by_path( $slug );

            if (
                $page instanceof WP_Post &&
                'publish' === get_post_status( $page )
            ) {
                return get_permalink( $page );
            }
        }

        return '';
    }
}

    if ( empty( $privacy_url ) ) {
        $privacy_url = home_url( '/gizlilik-politikasi/' );
    }

    ?>
    <div id="ms-cerez-banner"
         role="region"
         aria-label="Çerez bildirimi"
         style="display:none;position:fixed;bottom:0;left:0;right:0;z-index:99999;background-color:#111111;color:#e0e0e0;padding:12px 16px;flex-wrap:wrap;align-items:center;justify-content:space-between;gap:12px;border-top:1px solid #333333;font-family:Arial,Helvetica,sans-serif;font-size:14px;line-height:1.5;box-sizing:border-box;">

        <p style="margin:0;flex:1 1 220px;max-width:80ch;color:#e0e0e0;">
            Bu site, deneyiminizi iyileştirmek için çerezler kullanmaktadır.
            Kişisel verilerinizin işlenmesine ilişkin detaylar için
            <a href="<?php echo esc_url( $privacy_url ); ?>"
               style="color:#ffb800;text-decoration:underline;font-weight:600;">Gizlilik Politikamızı</a>
            inceleyebilirsiniz.
        </p>

        <div style="display:flex;flex-wrap:wrap;gap:8px;flex:0 0 auto;">

            <button id="ms-cerez-accept"
                    type="button"
                    style="cursor:pointer;border:none;border-radius:999px;padding:10px 18px;min-height:44px;background-color:#ffffff;color:#000000;font-size:14px;font-weight:600;white-space:nowrap;">
                Kabul Et
            </button>

            <button id="ms-cerez-reject"
                    type="button"
                    style="cursor:pointer;border:1px solid #ffffff;border-radius:999px;padding:10px 18px;min-height:44px;background-color:transparent;color:#ffffff;font-size:14px;font-weight:600;white-space:nowrap;">
                Reddet
            </button>

        </div>
    </div>

    <script>
    (function () {

        var STORAGE_KEY = 'kvkk_cookie_consent';

        function init() {
            var banner = document.getElementById('ms-cerez-banner');

            if (!banner) {
                return;
            }

            var stored = null;

            try {
                stored = window.localStorage.getItem(STORAGE_KEY);
            } catch (e) {
                stored = null;
            }

            if (stored) {
                return;
            }

            banner.style.display = 'flex';

            function choose(value) {
                try {
                    window.localStorage.setItem(STORAGE_KEY, value);
                } catch (e) {
                    // localStorage kullanılamıyorsa sessizce devam et
                }

                banner.style.display = 'none';
            }

            var acceptBtn = document.getElementById('ms-cerez-accept');
            var rejectBtn = document.getElementById('ms-cerez-reject');

            if (acceptBtn) {
                acceptBtn.addEventListener('click', function () {
                    choose('accepted');
                });
            }

            if (rejectBtn) {
                rejectBtn.addEventListener('click', function () {
                    choose('rejected');
                });
            }
        }

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }

    })();
    </script>
    <?php

}, 100 );
