/* ============================================================
   BODY CLASS
   ============================================================ */

add_filter(
    'body_class',
    function ( $classes ) {

        if ( is_singular( 'post' ) ) {
            $classes[] = 'mdg-blog-single-v1';
        }

        return $classes;
    }
);


/* ============================================================
   TEKİL BLOG YAZISI HTML
   ============================================================ */

add_filter(
    'the_content',
    function ( $content ) {

        if (
            is_admin() ||
            ! is_singular( 'post' ) ||
            ! in_the_loop() ||
            ! is_main_query()
        ) {
            return $content;
        }

        $post_id = get_the_ID();

        $title = get_the_title( $post_id );

        $blog_url = home_url( '/madagaskar-blog/' );

        $cities_url = home_url( '/sehirler/' );

        $ticket_url = home_url( '/bilet-al/' );

        $image = get_the_post_thumbnail_url(
            $post_id,
            'full'
        );

        $categories = get_the_category(
            $post_id
        );

        $category_name =
            ! empty( $categories )
                ? $categories[0]->name
                : 'Madagaskar Sirki Blog';

        $date = get_the_date(
            'd.m.Y',
            $post_id
        );

        ob_start();
        ?>

        <main id="mdg-blog-single">

            <section class="mdgbs-hero">

                <div class="mdgbs-wrap">

                    <a
                        href="<?php echo esc_url( $blog_url ); ?>"
                        class="mdgbs-back"
                    >
                        ← Madagaskar Blog
                    </a>

                    <div class="mdgbs-hero-card">

                        <div class="mdgbs-meta">

                            <span>
                                <?php echo esc_html( $category_name ); ?>
                            </span>

                            <time>
                                <?php echo esc_html( $date ); ?>
                            </time>

                        </div>

                        <h1>
                            <?php echo esc_html( $title ); ?>
                        </h1>

                    </div>

                </div>

            </section>


            <?php if ( $image ) : ?>

                <section class="mdgbs-image-section">

                    <div class="mdgbs-wrap">

                        <div class="mdgbs-featured">

                            <img
                                src="<?php echo esc_url( $image ); ?>"
                                alt="<?php echo esc_attr( $title ); ?>"
                            >

                        </div>

                    </div>

                </section>

            <?php endif; ?>


            <section class="mdgbs-content-section">

                <div class="mdgbs-wrap">

                    <article class="mdgbs-article">

                        <?php echo $content; ?>

                    </article>

                </div>

            </section>


            <section class="mdgbs-bottom">

                <div class="mdgbs-wrap">

                    <div class="mdgbs-cta">

                        <div>

                            <span>
                                MADAGASKAR SİRKİ TÜRKİYE
                            </span>

                            <h2>
                                Gösteriyi canlı keşfedin.
                            </h2>

                            <p>
                                Turne şehirlerini, güncel seansları
                                ve bilet seçeneklerini inceleyin.
                            </p>

                        </div>

                        <div class="mdgbs-buttons">

                            <a
                                href="<?php echo esc_url( $cities_url ); ?>"
                                class="mdgbs-btn mdgbs-red"
                            >
                                Turne Şehirleri
                            </a>

                            <a
                                href="<?php echo esc_url( $ticket_url ); ?>"
                                class="mdgbs-btn mdgbs-white"
                            >
                                Bilet Al
                            </a>

                        </div>

                    </div>


                    <a
                        href="<?php echo esc_url( $blog_url ); ?>"
                        class="mdgbs-blog-return"
                    >
                        ← Tüm blog yazılarına dön
                    </a>

                </div>

            </section>

        </main>

        <?php

        return ob_get_clean();

    },
    9999
);
