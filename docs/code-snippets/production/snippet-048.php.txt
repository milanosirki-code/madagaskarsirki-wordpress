function mdg_blog_v2_page_check() {

    return is_page(
        array(
            'madagaskar-blog',
            'blog'
        )
    );
}


/* ============================================================
   BODY CLASS
   ============================================================ */

    function ( $classes ) {

        if ( mdg_blog_v2_page_check() ) {
            $classes[] = 'mdg-blog-v2-page';

        return $classes;
    }
);

    'the_content',
    function ( $content ) {

        if (
            is_admin() ||
            ! mdg_blog_v2_page_check() ||

            ! is_main_query()
        ) {
            return $content;
        }

        return mdg_blog_v2_output();

    },
    9999
);
