/* ============================================================

   ============================================================ */

function mdg_is_corporate_page_v2() {
    return is_page( 'kurumsal' );
}


/* ============================================================
   2. BODY CLASS
   ============================================================ */

add_filter(
    'body_class',
    function ( $classes ) {

        if ( mdg_is_corporate_page_v2() ) {
            $classes[] = 'mdg-corporate-page';
        }

        return $classes;
    }
);

function mdg_corp_kommo_ready_v3() {

    if (
        ! defined( 'MS_KOMMO_TOKEN' ) ||
        ! defined( 'MS_KOMMO_BASE_URL' )
    ) {
        return false;
    }

    $token =
        trim(
            (string) MS_KOMMO_TOKEN
        );

    return (
        '' !== $token &&
        'YENI_KOMMO_TOKENINI_BURAYA_YAPISTIR' !== $token
    );
}


function mdg_corp_kommo_request_v3(
    $method,
    $path,
    $body = null
) {

    if ( ! mdg_corp_kommo_ready_v3() ) {

        return new WP_Error(
            'mdg_kommo_not_ready',
            'Kommo erişim ayarları hazır değil.'
        );
    }


    $url =
        rtrim(
            (string) MS_KOMMO_BASE_URL,
            '/'
        ) .
        '/' .
        ltrim(
            (string) $path,
            '/'
        );


    $args = array(
        'method'  => strtoupper( $method ),
        'timeout' => 20,
        'headers' => array(
            'Authorization' =>
                'Bearer ' . MS_KOMMO_TOKEN,
            'Accept' =>
                'application/json',
            'Content-Type' =>
                'application/json',
        ),
    );


    if ( null !== $body ) {
        $args['body'] =
            wp_json_encode(
                $body
            );
    }


    $response =
        wp_remote_request(
            $url,
            $args
        );


    if ( is_wp_error( $response ) ) {
        return $response;
    }


    $code =
        (int) wp_remote_retrieve_response_code(
            $response
        );


    if (
        $code < 200 ||
        $code >= 300
    ) {

        $response_body =
            wp_remote_retrieve_body(
                $response
            );

        return new WP_Error(
            'mdg_kommo_http_error',
            'Kommo HTTP ' . $code,
            array(
                'http_code' => $code,
                'response'  => $response_body,
            )
        );
    }


    if ( 204 === $code ) {
        return array();
    }


    $decoded =
        json_decode(
            wp_remote_retrieve_body(
                $response
            ),
            true
        );


    return is_array( $decoded )
        ? $decoded
        : array();
}


function mdg_corp_normalize_phone_v3( $phone ) {

    $digits =
        preg_replace(
            '/\D+/',
            '',
            (string) $phone
        );


    if ( ! $digits ) {
        return '';
    }


    /*
     * Türkiye:
     * 05xxxxxxxxx -> +905xxxxxxxxx
     * 5xxxxxxxxx  -> +905xxxxxxxxx
     * 905xxxxxxxxx -> +905xxxxxxxxx
     */

    if (
        11 === strlen( $digits ) &&
        '0' === substr( $digits, 0, 1 )
    ) {
        return
            '+90' .
            substr(
                $digits,
                1
            );
    }


    if (
        10 === strlen( $digits ) &&
        '5' === substr( $digits, 0, 1 )
    ) {
        return
            '+90' .
            $digits;
    }


    if (
        12 === strlen( $digits ) &&
        '90' === substr( $digits, 0, 2 )
    ) {
        return
            '+' .
            $digits;
    }


    return
        '+' .
        $digits;
}


function mdg_corp_extract_contact_phones_v3( $contact ) {

    $phones = array();

    $fields =
        isset(
            $contact['custom_fields_values']
        ) &&
        is_array(
            $contact['custom_fields_values']
        )
            ? $contact['custom_fields_values']
            : array();


    foreach ( $fields as $field ) {

        $field_code =
            isset( $field['field_code'] )
                ? strtoupper(
                    (string) $field['field_code']
                )
                : '';

        $values =
            isset( $field['values'] ) &&
            is_array( $field['values'] )
                ? $field['values']
                : array();


        /*
         * Bazı Kommo cevaplarında PHONE field_code gelmeyebilir.
         * Bu nedenle değerleri de güvenli şekilde tarıyoruz.
         */
        foreach ( $values as $value_row ) {

            if (
                ! isset(
                    $value_row['value']
                )
            ) {
                continue;
            }

            $value =
                (string) $value_row['value'];

            if (
                'PHONE' === $field_code ||
                preg_match(
                    '/[0-9]{7,}/',
                    $value
                )
            ) {

                $normalized =
                    mdg_corp_normalize_phone_v3(
                        $value
                    );

                if ( $normalized ) {
                    $phones[] =
                        $normalized;
                }
            }
        }
    }


    return array_values(
        array_unique(
            $phones
        )
    );
}


function mdg_corp_find_contact_v3( $phone ) {

    $normalized =
        mdg_corp_normalize_phone_v3(
            $phone
        );


    if ( ! $normalized ) {
        return 0;
    }


    $result =
        mdg_corp_kommo_request_v3(
            'GET',
            'contacts?query=' .
            rawurlencode(
                $normalized
            ) .
            '&limit=25'
        );


    if ( is_wp_error( $result ) ) {
        return $result;
    }


    $contacts =
        isset(
            $result['_embedded']['contacts']
        ) &&
        is_array(
            $result['_embedded']['contacts']
        )
            ? $result['_embedded']['contacts']
            : array();


    foreach ( $contacts as $contact ) {

        $phones =
            mdg_corp_extract_contact_phones_v3(
                $contact
            );

        if (
            in_array(
                $normalized,
                $phones,
                true
            )
        ) {
            return
                isset( $contact['id'] )
                    ? absint(
                        $contact['id']
                    )
                    : 0;
        }
    }


    return 0;
}


function mdg_corp_create_contact_v3(
    $contact_name,
    $phone
) {

    $normalized =
        mdg_corp_normalize_phone_v3(

    $body = array(

            'custom_fields_values' =>
                array(
                    array(
                        'field_code' =>
                            'PHONE',
                        'values' =>
                            array(
                                array(
                                    'value' =>
                                        $normalized,
                                    'enum_code' =>
                                        'MOB',
                                ),
                            ),
                    ),
                ),
        ),
    );


    $result =
        mdg_corp_kommo_request_v3(
            'POST',
            'contacts',
            $body
        );


    if ( is_wp_error( $result ) ) {
        return $result;
    }


    $contact_id =
        isset(
            $result['_embedded']['contacts'][0]['id']
        )
            ? absint(
                $result['_embedded']['contacts'][0]['id']
            )
            : 0;


    if ( ! $contact_id ) {

        return new WP_Error(
            'mdg_kommo_contact_missing',
            'Kommo kişi kaydı oluşturuldu ancak kişi ID alınamadı.'
        );

}

function mdg_corp_resolve_pipeline_status_v5() {

    $configured_pipeline_id =
        defined( 'MDG_CORP_KOMMO_PIPELINE_ID' )
            ? absint( MDG_CORP_KOMMO_PIPELINE_ID )
            : 0;

    $configured_status_id =
        defined( 'MDG_CORP_KOMMO_STATUS_ID' )
            ? absint( MDG_CORP_KOMMO_STATUS_ID )
            : 0;


    if (
        $configured_pipeline_id &&
        $configured_status_id
    ) {
        return array(
            'pipeline_id' => $configured_pipeline_id,
            'status_id'   => $configured_status_id,
            'source'      => 'configured_ids',
        );
    }


    $cached =
        get_transient(
            'mdg_corp_kommo_pipeline_status_v5'
        );

    if (
        is_array( $cached ) &&
        ! empty( $cached['pipeline_id'] ) &&
        ! empty( $cached['status_id'] )
    ) {
        return $cached;
    }


    $result =
        mdg_corp_kommo_request_v3(
            'GET',
            'leads/pipelines'
        );


    if ( is_wp_error( $result ) ) {
        return $result;
    }


    $pipelines =
        isset(
            $result['_embedded']['pipelines']
        ) &&
        is_array(
            $result['_embedded']['pipelines']
        )
            ? $result['_embedded']['pipelines']
            : array();


    $wanted_pipeline_name =
        defined( 'MDG_CORP_KOMMO_PIPELINE_NAME' )
            ? trim(
                (string) MDG_CORP_KOMMO_PIPELINE_NAME
            )
            : 'Kurumsal Talepler';

    $wanted_status_name =
        defined( 'MDG_CORP_KOMMO_STATUS_NAME' )
            ? trim(
                (string) MDG_CORP_KOMMO_STATUS_NAME
            )
            : 'İlk Temas';


    foreach ( $pipelines as $pipeline ) {

        $pipeline_name =
            isset( $pipeline['name'] )
                ? trim(
                    (string) $pipeline['name']
                )
                : '';

        if (
            0 !== strcasecmp(
                $pipeline_name,
                $wanted_pipeline_name
            )
        ) {
            continue;
        }


        $pipeline_id =
            isset( $pipeline['id'] )
                ? absint(
                    $pipeline['id']
                )
                : 0;


        $statuses =
            isset(
                $pipeline['_embedded']['statuses']
            ) &&
            is_array(
                $pipeline['_embedded']['statuses']
            )
                ? $pipeline['_embedded']['statuses']
                : array();


        foreach ( $statuses as $status ) {

            $status_name =
                isset( $status['name'] )
                    ? trim(
                        (string) $status['name']
                    )
                    : '';

            if (
                0 !== strcasecmp(
                    $status_name,
                    $wanted_status_name
                )
            ) {
                continue;
            }


            $status_id =
                isset( $status['id'] )
                    ? absint(
                        $status['id']
                    )
                    : 0;


            if (
                $pipeline_id &&
                $status_id
            ) {

                $resolved =
                    array(
                        'pipeline_id' =>
                            $pipeline_id,
                        'status_id' =>
                            $status_id,
                        'source' =>
                            'auto_name_match',
                    );


                set_transient(
                    'mdg_corp_kommo_pipeline_status_v5',
                    $resolved,
                    6 * HOUR_IN_SECONDS
                );


                return $resolved;
            }
        }


        return new WP_Error(
            'mdg_kommo_status_not_found',
            sprintf(
                'Kommo pipeline bulundu ancak "%s" aşaması bulunamadı.',
                $wanted_status_name
            )
        );
    }


    return new WP_Error(
        'mdg_kommo_pipeline_not_found',
        sprintf(
            'Kommo "%s" pipeline bulunamadı.',
            $wanted_pipeline_name
        )
    );
}

function mdg_corp_get_all_lead_fields_v6() {

    $all_fields = array();

    for ( $page = 1; $page <= 20; $page++ ) {

        $result =
            mdg_corp_kommo_request_v3(
                'GET',
                'leads/custom_fields?limit=250&page=' .
                absint( $page )
            );

        if ( is_wp_error( $result ) ) {
            return $result;
        }


        $fields =
            isset(
                $result['_embedded']['custom_fields']
            ) &&
            is_array(
                $result['_embedded']['custom_fields']
            )
                ? $result['_embedded']['custom_fields']
                : array();


        if ( empty( $fields ) ) {
            break;
        }


        $all_fields =
            array_merge(
                $all_fields,
                $fields
            );


        $page_count =
            isset( $result['_page_count'] )
                ? absint( $result['_page_count'] )
                : 0;

        if (
            $page_count &&
            $page >= $page_count
        ) {
            break;
        }


        if ( count( $fields ) < 250 ) {
            break;
        }
    }


    return $all_fields;
}


function mdg_corp_resolve_lead_fields_v6() {

    $cache_key =
        'mdg_corp_kommo_lead_fields_v6';


    $cached =
        get_transient(
            $cache_key
        );


    $specs =
        mdg_corp_required_lead_fields_v6();


    if ( is_array( $cached ) ) {

        $all_present = true;

        foreach ( array_keys( $specs ) as $key ) {

            if ( empty( $cached[ $key ] ) ) {
                $all_present = false;
                break;
            }
        }


        if ( $all_present ) {
            return $cached;
        }
    }


    $existing_fields =
        mdg_corp_get_all_lead_fields_v6();


    if ( is_wp_error( $existing_fields ) ) {
        return $existing_fields;
    }


    $resolved = array();
    $missing  = array();


    foreach ( $specs as $key => $spec ) {

        $wanted_name =
            trim(
                (string) $spec['name']
            );


        foreach ( $existing_fields as $field ) {

            $field_name =
                isset( $field['name'] )
                    ? trim(
                        (string) $field['name']
                    )
                    : '';

            $field_id =
                isset( $field['id'] )
                    ? absint(
                        $field['id']
                    )
                    : 0;


            if (
                $field_id &&
                0 === strcasecmp(
                    $field_name,
                    $wanted_name
                )
            ) {

                $resolved[ $key ] =
                    $field_id;

                break;
            }
        }


        if ( empty( $resolved[ $key ] ) ) {

            $missing[ $key ] =
                array(
                    'name' =>
                        $wanted_name,
                    'type' =>
                        (string) $spec['type'],
                );
        }
    }


    /*
     * Eksik alanları Kommo'da otomatik oluştur.
     * Bu endpoint Kommo tarafında admin yetkisi ister.
     * Yetki yoksa lead yine oluşturulur; ayrıntılar notta kalır
     * ve WordPress kaydında "Kısmi" durumu görülebilir.
     */
    if ( ! empty( $missing ) ) {

        $create_payload =
            array_values(
                $missing
            );


        $created =
            mdg_corp_kommo_request_v3(
                'POST',
                'leads/custom_fields',
                $create_payload
            );


        if ( is_wp_error( $created ) ) {
            return $created;
        }


        $created_fields =
            isset(
                $created['_embedded']['custom_fields']
            ) &&
            is_array(
                $created['_embedded']['custom_fields']
            )
                ? $created['_embedded']['custom_fields']
                : array();


        foreach ( $missing as $key => $spec ) {

            foreach ( $created_fields as $field ) {

                $field_name =
                    isset( $field['name'] )
                        ? trim(
                            (string) $field['name']
                        )
                        : '';

                $field_id =
                    isset( $field['id'] )
                        ? absint(
                            $field['id']
                        )
                        : 0;


                if (
                    $field_id &&
                    0 === strcasecmp(
                        $field_name,
                        (string) $spec['name']
                    )
                ) {

                    $resolved[ $key ] =
                        $field_id;

                    break;
                }
            }
        }
    }


    foreach ( array_keys( $specs ) as $key ) {

        if ( empty( $resolved[ $key ] ) ) {

            return new WP_Error(
                'mdg_kommo_field_missing',
                'Kommo özel alanı çözülemedi: ' .
                $specs[ $key ]['name']
            );
        }
    }


    set_transient(
        $cache_key,
        $resolved,
        6 * HOUR_IN_SECONDS
    );


    return $resolved;
}


function mdg_corp_build_custom_field_values_v6(
    $field_ids,
    $data
) {

    $values = array();


    foreach ( $field_ids as $key => $field_id ) {

        if (
            ! array_key_exists(
                $key,
                $data
            )
        ) {
            continue;
        }


        $value =
            $data[ $key ];


        if (
            null === $value ||
            '' === trim(
                (string) $value
            )
        ) {
            continue;
        }


        if ( 'audience' === $key ) {

            $value =
                (string) absint(
                    $value
                );

            if ( '0' === $value ) {
                continue;
            }
        }


        $values[] =
            array(
                'field_id' =>
                    absint(
                        $field_id
                    ),
                'values' =>
                    array(
                        array(
                            'value' =>
                                (string) $value,
                        ),
                    ),
            );
    }


    return $values;
}


function mdg_corp_update_lead_fields_v6(
    $kommo_lead_id,
    $data
) {

    $field_ids =
        mdg_corp_resolve_lead_fields_v6();


    if ( is_wp_error( $field_ids ) ) {
        return $field_ids;
    }


    $custom_fields_values =
        mdg_corp_build_custom_field_values_v6(
            $field_ids,
            $data
        );


    if ( empty( $custom_fields_values ) ) {
        return $field_ids;
    }


    $result =
        mdg_corp_kommo_request_v3(
            'PATCH',
            'leads/' .
            absint(
                $kommo_lead_id
            ),
            array(
                'custom_fields_values' =>
                    $custom_fields_values,
            )
        );


    if ( is_wp_error( $result ) ) {
        return $result;
    }


    return $field_ids;
}


function mdg_corp_create_lead_v3(
    $company,
    $city,
    $event_type,
    $contact_id
) {

    $lead =
        array(
            'name' =>
                sprintf(
                    'Kurumsal Talep — %s — %s',
                    $company,
                    $city
                ),
            '_embedded' =>
                array(
                    'contacts' =>
                        array(
                            array(
                                'id' =>
                                    absint(
                                        $contact_id
                                    ),
                            ),
                        ),
                    'tags' =>
                        array(
                            array(
                                'name' =>
                                    MDG_CORP_KOMMO_TAG,
                            ),
                            array(
                                'name' =>
                                    $event_type,
                            ),
                        ),
                ),
        );


    $routing =
        mdg_corp_resolve_pipeline_status_v5();


    if ( is_wp_error( $routing ) ) {
        return $routing;
    }


    $lead['pipeline_id'] =
        absint(
            $routing['pipeline_id']
        );

    $lead['status_id'] =
        absint(
            $routing['status_id']
        );


    $result =
        mdg_corp_kommo_request_v3(
            'POST',
            'leads',
            array(
                $lead,
            )
        );


    if ( is_wp_error( $result ) ) {
        return $result;
    }


    $lead_id =
        isset(
            $result['_embedded']['leads'][0]['id']
        )
            ? absint(
                $result['_embedded']['leads'][0]['id']
            )
            : 0;


    if ( ! $lead_id ) {

        return new WP_Error(
            'mdg_kommo_lead_missing',
            'Kommo lead oluşturuldu ancak lead ID alınamadı.'
        );
    }


    return $lead_id;
}

function mdg_corp_sync_to_kommo_v3(
    $local_lead_id,
    $data
) {

    if ( ! mdg_corp_kommo_ready_v3() ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'config_missing'
        );

        return new WP_Error(
            'mdg_kommo_not_ready',
            'Kommo ayarları bulunamadı.'
        );
    }


    $existing_lead =
        absint(
            get_post_meta(
                $local_lead_id,
                '_mdg_kommo_lead_id',
                true
            )
        );


    if ( $existing_lead ) {
        return $existing_lead;
    }


    update_post_meta(
        $local_lead_id,
        '_mdg_kommo_status',
        'syncing'
    );


    $contact_id =
        mdg_corp_get_or_create_contact_v3(
            $data['contact_name'],
            $data['phone']
        );


    if ( is_wp_error( $contact_id ) ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'error'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_error',
            $contact_id->get_error_message()
        );

        return $contact_id;
    }


    update_post_meta(
        $local_lead_id,
        '_mdg_kommo_contact_id',
        absint(
            $contact_id
        )
    );


    $kommo_lead_id =
        mdg_corp_create_lead_v3(
            $data['company'],
            $data['city'],
            $data['event_type'],
            $contact_id
        );


    if ( is_wp_error( $kommo_lead_id ) ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'error'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_error',
            $kommo_lead_id->get_error_message()
        );

        return $kommo_lead_id;
    }


    update_post_meta(
        $local_lead_id,
        '_mdg_kommo_lead_id',
        absint(
            $kommo_lead_id
        )
    );


    $routing =
        mdg_corp_resolve_pipeline_status_v5();

    if ( is_array( $routing ) ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_pipeline_id',
            absint( $routing['pipeline_id'] )
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status_id',
            absint( $routing['status_id'] )
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_routing_source',
            sanitize_text_field(
                isset( $routing['source'] )
                    ? $routing['source']
                    : ''
            )
        );
    }


    /*
     * Kurumsal bilgileri Kommo lead özel alanlarına yaz.
     * Not kaydı ayrıca tutulmaya devam eder; böylece iki ayrı
     * okunabilir kayıt vardır.
     */
    $field_result =
        mdg_corp_update_lead_fields_v6(
            $kommo_lead_id,
            $data
        );


    if ( is_wp_error( $field_result ) ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_status',
            'error'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_error',
            $field_result->get_error_message()
        );

    } else {

        delete_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_error'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_status',
            'success'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_ids',
            wp_json_encode(
                $field_result
            )
        );
    }


    $note_result =
        mdg_corp_add_lead_note_v3(
            $kommo_lead_id,
            $local_lead_id,
            $data
        );


    if ( is_wp_error( $note_result ) ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'partial'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_error',
            'Lead oluşturuldu; not eklenemedi: ' .
            $note_result->get_error_message()
        );

        return $kommo_lead_id;
    }


    $field_status =
        (string) get_post_meta(
            $local_lead_id,
            '_mdg_kommo_field_status',
            true
        );


    if ( 'error' === $field_status ) {

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'partial'
        );

        $field_error =
            (string) get_post_meta(
                $local_lead_id,
                '_mdg_kommo_field_error',
                true
            );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_error',
            'Lead ve not oluşturuldu; özel alanlar yazılamadı: ' .
            $field_error
        );

    } else {

        delete_post_meta(
            $local_lead_id,
            '_mdg_kommo_error'
        );

        update_post_meta(
            $local_lead_id,
            '_mdg_kommo_status',
            'success'
        );
    }


    update_post_meta(
        $local_lead_id,
        '_mdg_kommo_synced_at',
        current_time(
            'mysql'
        )
    );


    return $kommo_lead_id;
}

function mdg_handle_corporate_quote_v2() {

    $redirect_base =
        home_url( '/kurumsal/' );


    if (
        ! isset( $_POST['mdg_corporate_quote_nonce'] ) ||
        ! wp_verify_nonce(
            sanitize_text_field(
                wp_unslash(
                    $_POST['mdg_corporate_quote_nonce']
                )
            ),
            'mdg_corporate_quote'
        )
    ) {

        wp_safe_redirect(
            add_query_arg(
                'mdg_teklif',
                'error',
                $redirect_base
            ) . '#kurumsal-form'
        );

        exit;
    }


    /* Basit bot kontrolü */

    if (
        ! empty( $_POST['website'] )
    ) {

        wp_safe_redirect(
            add_query_arg(
                'mdg_teklif',
                'ok',
                $redirect_base
            ) . '#kurumsal-form'
        );

        exit;
    }


    /* Çok hızlı tekrar gönderimi engelle */

    $ip =
        isset( $_SERVER['REMOTE_ADDR'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_SERVER['REMOTE_ADDR']
                )
            )
            : 'unknown';

    $rate_key =
        'mdg_corp_quote_' .
        md5( $ip );

    if ( get_transient( $rate_key ) ) {

        wp_safe_redirect(
            add_query_arg(
                'mdg_teklif',
                'rate',
                $redirect_base
            ) . '#kurumsal-form'
        );

        exit;
    }


    $company =
        isset( $_POST['company'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['company']
                )
            )
            : '';

    $contact_name =
        isset( $_POST['contact_name'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['contact_name']
                )
            )
            : '';

    $phone =
        isset( $_POST['phone'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['phone']
                )
            )
            : '';

    $city =
        isset( $_POST['city'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['city']
                )
            )
            : '';

    $event_date =
        isset( $_POST['event_date'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['event_date']
                )
            )
            : '';

    $event_type =
        isset( $_POST['event_type'] )
            ? sanitize_text_field(
                wp_unslash(
                    $_POST['event_type']
                )
            )
            : '';

    $audience =
        isset( $_POST['audience'] )
            ? absint(
                $_POST['audience']
            )
            : 0;

    $note =
        isset( $_POST['note'] )
            ? sanitize_textarea_field(
                wp_unslash(
                    $_POST['note']
                )
            )
            : '';

    $consent =
        ! empty(
            $_POST['contact_consent']
        );


    $allowed_types =
        array(
            'Okul',
            'Belediye',
            'AVM',
            'Festival',
            'Özel Organizasyon',
            'Diğer',
        );


    if (
        ! $company ||
        ! $contact_name ||
        ! $phone ||
        ! $city ||
        ! $event_type ||
        ! in_array(
            $event_type,
            $allowed_types,
            true
        ) ||
        ! $consent
    ) {

        wp_safe_redirect(
            add_query_arg(
                'mdg_teklif',
                'error',
                $redirect_base
            ) . '#kurumsal-form'
        );

        exit;
    }


    $post_title =
        sprintf(
            '%s — %s — %s',
            $company,
            $city,
            current_time( 'd.m.Y H:i' )
        );


    $lead_id =
        wp_insert_post(
            array(
                'post_type'   => 'mdg_corp_lead',
                'post_status' => 'private',
                'post_title'  => $post_title,
            ),
            true
        );


    if ( is_wp_error( $lead_id ) ) {

        wp_safe_redirect(
            add_query_arg(
                'mdg_teklif',
                'error',
                $redirect_base
            ) . '#kurumsal-form'
        );

        exit;
    }


    update_post_meta(
        $lead_id,
        '_mdg_company',
        $company
    );

    update_post_meta(
        $lead_id,
        '_mdg_contact_name',
        $contact_name
    );

    update_post_meta(
        $lead_id,
        '_mdg_phone',
        $phone
    );

    update_post_meta(
        $lead_id,
        '_mdg_city',
        $city
    );

    update_post_meta(
        $lead_id,
        '_mdg_event_date',
        $event_date
    );

    update_post_meta(
        $lead_id,
        '_mdg_event_type',
        $event_type
    );

    update_post_meta(
        $lead_id,
        '_mdg_audience',
        $audience
    );

    update_post_meta(
        $lead_id,
        '_mdg_note',
        $note
    );

    update_post_meta(
        $lead_id,
        '_mdg_contact_consent',
        1
    );


    /*
     * Kommo senkronizasyonu.
     * Kommo geçici olarak erişilemez olsa bile WordPress kaydı korunur
     * ve ziyaretçiye form hatası gösterilmez.
     */

    $kommo_data =
        array(
            'company' =>
                $company,
            'contact_name' =>
                $contact_name,
            'phone' =>
                $phone,
            'city' =>
                $city,
            'event_date' =>
                $event_date,
            'event_type' =>
                $event_type,
            'audience' =>
                $audience,
            'note' =>
                $note,
        );


    mdg_corp_sync_to_kommo_v3(
        $lead_id,
        $kommo_data
    );


    /* Site yöneticisine e-posta bildirimi */

    $recipient =
        apply_filters(
            'mdg_corporate_quote_recipient',
            get_option( 'admin_email' )
        );


    $subject =
        sprintf(
            'Yeni Kurumsal Teklif Talebi — %s',
            $company
        );


    $message =
        "Yeni kurumsal teklif talebi alındı.\n\n" .
        "Kurum: {$company}\n" .
        "Yetkili: {$contact_name}\n" .
        "Telefon: {$phone}\n" .
        "Şehir: {$city}\n" .
        "Planlanan Tarih: {$event_date}\n" .
        "Etkinlik Türü: {$event_type}\n" .
        "Tahmini Katılımcı: " .
        ( $audience ? $audience : '-' ) .
        "\n\nNot:\n" .
        ( $note ? $note : '-' ) .
        "\n\nTalep No: {$lead_id}\n";


    if ( is_email( $recipient ) ) {

        wp_mail(
            $recipient,
            $subject,
            $message
        );
    }


    set_transient(
        $rate_key,
        1,
        30
    );


    wp_safe_redirect(
        add_query_arg(
            'mdg_teklif',
            'ok',
            $redirect_base
        ) . '#kurumsal-form'
    );

    exit;
}

);

            }
        }
