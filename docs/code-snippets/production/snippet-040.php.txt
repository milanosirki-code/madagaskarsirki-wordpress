/* ============================================================
   3. MEVCUT İÇERİKTEN FOTOĞRAFLARI ÇIKAR
   ============================================================ */

function mdg_extract_gallery_images_v1( $content ) {

    $images = array();

    /*
     * Mevcut sayfa içerisindeki img etiketlerini bul.
     */
    preg_match_all(
        '/<img[^>]+>/i',
        $content,
        $img_tags
    );

    if ( empty( $img_tags[0] ) ) {
        return $images;
    }


    foreach ( $img_tags[0] as $tag ) {

        $src = '';
        $alt = 'Madagaskar Sirki Gösteri Fotoğrafı';


        /*
         * SRC
         */
        if (
            preg_match(
                '/src=["\']([^"\']+)["\']/i',
                $tag,
                $src_match
            )
        ) {
            $src = html_entity_decode(
                $src_match[1]
            );
        }


        /*
         * ALT
         */
        if (
            preg_match(
                '/alt=["\']([^"\']*)["\']/i',
                $tag,
                $alt_match
            )
        ) {

            if ( ! empty( trim( $alt_match[1] ) ) ) {

                $alt = wp_strip_all_tags(
                    html_entity_decode(
                        $alt_match[1]
                    )
                );
            }
        }


        if ( empty( $src ) ) {
            continue;
        }


        /*
         * Aynı görseli ikinci kez ekleme.
         */
        $already_exists = false;

        foreach ( $images as $existing ) {

            if ( $existing['src'] === $src ) {
                $already_exists = true;
                break;
            }
        }

        if ( $already_exists ) {
            continue;
        }


        $images[] = array(
            'src' => esc_url_raw( $src ),
            'alt' => $alt,
        );
    }


    return $images;
}

/* ============================================================
   5. MEVCUT GALERİ İÇERİĞİNİ DEĞİŞTİR
   ============================================================ */

add_filter(
    'the_content',
    function ( $content ) {

        if (
            is_admin() ||
            ! mdg_is_gallery_page_v1() ||
            ! in_the_loop() ||
            ! is_main_query()
        ) {
            return $content;
        }


        /*
         * Önce eski sayfadaki fotoğrafları yakala.
         */
        $images =
            mdg_extract_gallery_images_v1(
                $content
            );


        /*
         * Sonra eski tasarımı tamamen değiştir.
         */
        return
            mdg_render_gallery_v1(
                $images
            );

    },
    999
);

/* ============================================================
   7. LIGHTBOX JS
   ============================================================ */

add_action(
    'wp_footer',
    function () {

        if ( ! mdg_is_gallery_page_v1() ) {
            return;
        }

        ?>

<script id="mdg-gallery-lightbox-v1">

document.addEventListener(
    'DOMContentLoaded',
    function () {

        const gallery =
            document.getElementById(
                'mdg-gallery'
            );

        if ( ! gallery ) {
            return;
        }


        const lightbox =
            document.getElementById(
                'mdgg-lightbox'
            );


        if ( ! lightbox ) {
            return;
        }


        const lightboxImage =
            lightbox.querySelector('img');


        const closeButton =
            lightbox.querySelector(
                '.mdgg-lightbox-close'
            );


        const photos =
            gallery.querySelectorAll(
                '[data-mdgg-image]'
            );


        function openLightbox( src, alt ) {

            lightboxImage.src =
                src;

            lightboxImage.alt =
                alt || 'Madagaskar Sirki Galeri';


            lightbox.classList.add(
                'is-open'
            );


            lightbox.setAttribute(
                'aria-hidden',
                'false'
            );


            document.documentElement.style.overflow =
                'hidden';
        }


        function closeLightbox() {

            lightbox.classList.remove(
                'is-open'
            );


            lightbox.setAttribute(
                'aria-hidden',
                'true'
            );


            lightboxImage.src =
                '';


            document.documentElement.style.overflow =
                '';
        }


        photos.forEach(
            function ( photo ) {

                photo.addEventListener(
                    'click',
                    function () {

                        openLightbox(
                            photo.dataset.mdggImage,
                            photo.getAttribute(
                                'aria-label'
                            )
                        );
                    }
                );
            }
        );


        closeButton.addEventListener(
            'click',
            closeLightbox
        );


        lightbox.addEventListener(
            'click',
            function ( event ) {

                if (
                    event.target === lightbox
                ) {
                    closeLightbox();
                }
            }
        );


        document.addEventListener(
            'keydown',
            function ( event ) {

                if (
                    event.key === 'Escape' &&
                    lightbox.classList.contains(
                        'is-open'
                    )
                ) {
                    closeLightbox();
                }
            }
        );

    }
);

</script>

        <?php
    },
    9999
);
