/* ============================================================
   GÜVENLİ SAYFA KONTROLÜ
   ============================================================ */

function mdg_social_share_allowed() {

    if ( is_admin() ) {
        return false;
    }

    /* WooCommerce özel sayfaları */
    if ( function_exists( 'is_cart' ) && is_cart() ) {
        return false;
    }

    if ( function_exists( 'is_checkout' ) && is_checkout() ) {
        return false;
    }

    if ( function_exists( 'is_account_page' ) && is_account_page() ) {
        return false;
    }

    /* Özel / hassas Madagaskar sayfaları */
    if (
        is_page(
            array(
                'biletlerim',
                'bilet-al'
            )
        )
    ) {
        return false;
    }

    /* URL'de sipariş/token varsa gösterme */
    if (
        isset( $_GET['order_id'] ) ||
        isset( $_GET['token'] ) ||
        isset( $_GET['key'] )
    ) {
        return false;
    }

    return is_singular();
}


/* ============================================================
   PAYLAŞIM HTML
   ============================================================ */

function mdg_social_share_html() {

    $url =
        get_permalink();

    $title =
        get_the_title();

    $wa =
        'https://wa.me/?text=' .
        rawurlencode(
            $title . ' ' . $url
        );

    $facebook =
        'https://www.facebook.com/sharer/sharer.php?u=' .
        rawurlencode( $url );

    $x =
        'https://twitter.com/intent/tweet?text=' .
        rawurlencode( $title ) .
        '&url=' .
        rawurlencode( $url );

    $instagram =
        'https://www.instagram.com/madagaskarsirkiturkiye/';

    ob_start();
    ?>

    <section class="mdg-social-share">

        <div class="mdg-social-share-inner">

            <div class="mdg-social-title">

                <span>
                    MADAGASKAR SİRKİ
                </span>

                <strong>
                    Bu sayfayı paylaş
                </strong>

            </div>


            <div class="mdg-social-buttons">


                <a
                    href="<?php echo esc_url( $wa ); ?>"
                    target="_blank"
                    rel="noopener noreferrer"
                    class="mdg-social-btn mdg-social-wa"
                >
                    WhatsApp
                </a>


                <a
                    href="<?php echo esc_url( $facebook ); ?>"
                    target="_blank"
                    rel="noopener noreferrer"
                    class="mdg-social-btn"
                >
                    Facebook
                </a>


                <a
                    href="<?php echo esc_url( $x ); ?>"
                    target="_blank"
                    rel="noopener noreferrer"
                    class="mdg-social-btn"
                >
                    X
                </a>


                <button
                    type="button"
                    class="mdg-social-btn mdg-copy-link"
                    data-url="<?php echo esc_url( $url ); ?>"
                >
                    Linki Kopyala
                </button>


                <a
                    href="<?php echo esc_url( $instagram ); ?>"
                    target="_blank"
                    rel="noopener noreferrer"
                    class="mdg-social-btn mdg-social-instagram"
                >
                    Instagram
                </a>


            </div>

        </div>

    </section>

    <?php

    return ob_get_clean();
}


/* ============================================================
   İÇERİKTEN SONRA EKLE
   ============================================================ */

add_filter(
    'the_content',
    function ( $content ) {

        if (
            ! mdg_social_share_allowed() ||
            ! in_the_loop() ||
            ! is_main_query()
        ) {
            return $content;
        }

        return
            $content .
            mdg_social_share_html();

    },
    9000
);

/* ============================================================
   LINK KOPYALA
   ============================================================ */

add_action(
    'wp_footer',
    function () {

        if ( ! mdg_social_share_allowed() ) {
            return;
        }

        ?>

<script>

document.addEventListener(
    'click',
    function ( event ) {

        const button =
            event.target.closest(
                '.mdg-copy-link'
            );

        if ( ! button ) {
            return;
        }

        const url =
            button.dataset.url;

        if (
            navigator.clipboard &&
            url
        ) {

            navigator.clipboard
                .writeText( url )
                .then(
                    function () {

                        const oldText =
                            button.textContent;

                        button.textContent =
                            'Kopyalandı ✓';

                        setTimeout(
                            function () {

                                button.textContent =
                                    oldText;

                            },
                            1800
                        );
                    }
                );
        }

    }
);

</script>

        <?php
    },
    9999
);
