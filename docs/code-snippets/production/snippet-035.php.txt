 * =========================================================
 *
 * Şehir kaydında ham tarih alanı varsa onu, yoksa
 * "2 Ekim 2026" biçimindeki tarih etiketini okur.
 * Tarih anlaşılamazsa etiketi göstermez.

if ( ! function_exists( 'ms_home_v3_city_is_today' ) ) {

	function ms_home_v3_city_is_today( $city ) {

		if ( ! is_array( $city ) ) {
			return false;
		}

		$today = function_exists( 'wp_date' )
			? wp_date( 'Y-m-d' )
			: date( 'Y-m-d' );

		$keys = array(
			'date',
			'date_raw',
			'event_date',
			'start_date',
			'date_ymd',
			'ymd',
		);

		foreach ( $keys as $key ) {

			if ( empty( $city[ $key ] ) || ! is_scalar( $city[ $key ] ) ) {
				continue;
			}

			$value = (string) $city[ $key ];

			if ( preg_match( '/^(\d{4})-(\d{2})-(\d{2})/', $value, $m ) ) {
				return ( $m[1] . '-' . $m[2] . '-' . $m[3] ) === $today;
			}
		}

		if ( empty( $city['date_label'] ) || ! is_scalar( $city['date_label'] ) ) {
			return false;
		}

		$label = (string) $city['date_label'];

		if ( ! preg_match( '/(\d{1,2})\s+([^\s\d]+)\s+(\d{4})/u', $label, $m ) ) {
			return false;
		}

		$month_name = strtr(
			$m[2],
			array(
				'İ' => 'i',
				'I' => 'ı',
			)
		);

		$month_name = function_exists( 'mb_strtolower' )
			? mb_strtolower( $month_name, 'UTF-8' )
			: strtolower( $month_name );

		$months = array(
			'ocak'    => 1,
			'şubat'   => 2,
			'mart'    => 3,
			'nisan'   => 4,
			'mayıs'   => 5,
			'haziran' => 6,
			'temmuz'  => 7,
			'ağustos' => 8,
			'eylül'   => 9,
			'ekim'    => 10,
			'kasım'   => 11,
			'aralık'  => 12,
		);

		if ( ! isset( $months[ $month_name ] ) ) {
			return false;
		}

		$date = sprintf(
			'%04d-%02d-%02d',
			(int) $m[3],
			$months[ $month_name ],
			(int) $m[1]
		);

		return $date === $today;
	}
}
