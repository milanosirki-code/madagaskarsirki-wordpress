/* ============================================================
   2. EKSİK GÖSTERİ SAYFALARINI OLUŞTUR
   ============================================================ */

add_action(
    'admin_init',
    function () {

        if ( ! current_user_can( 'manage_options' ) ) {
            return;
        }

        $parent = get_page_by_path(
            'gosteriler',
            OBJECT,
            'page'
        );

        if ( ! $parent instanceof WP_Post ) {
            return;
        }

        $shows = mdg_show_catalog_v2();

        foreach ( $shows as $slug => $show ) {

            $existing = get_page_by_path(
                'gosteriler/' . $slug,
                OBJECT,
                'page'
            );

            if ( $existing instanceof WP_Post ) {
                continue;
            }

            wp_insert_post(
                array(

                    'post_type'    => 'page',
                    'post_status'  => 'publish',

                    'post_title'   =>
                        $show['title'],

                    'post_name'    =>
                        $slug,

                    'post_parent'  =>
                        $parent->ID,

                    'post_content' =>
                        '',

                )
            );
        }

    },
    20
);



/* ============================================================
   3. AKTİF GÖSTERİ SAYFASINI BUL
   ============================================================ */

function mdg_current_show_v2() {

    if ( ! is_page() ) {
        return false;
    }

    global $post;

    if ( ! $post instanceof WP_Post ) {
        return false;
    }

    $shows = mdg_show_catalog_v2();

    $slug = sanitize_title(
        $post->post_name
    );

    if ( ! isset( $shows[ $slug ] ) ) {
        return false;
    }


    /*
     * Gösteriler üst sayfasının altında olduğundan emin ol.
     */
    if ( $post->post_parent ) {

        $parent = get_post(
            $post->post_parent
        );

        if (
            ! $parent instanceof WP_Post ||
            'gosteriler' !== $parent->post_name
        ) {
            return false;
        }
    }

    return $shows[ $slug ];
}



/* ============================================================
   4. BODY CLASS
   ============================================================ */

add_filter(
    'body_class',
    function ( $classes ) {

        if ( mdg_current_show_v2() ) {

            $classes[] =
                'mdg-show-detail-page';
        }

        return $classes;
    }
);
