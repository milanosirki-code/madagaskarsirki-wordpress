add_action( 'wp_head', function () {

    if ( is_admin() || ! is_singular( 'product' ) ) {
        return;
    }

    global $post;

    if ( ! $post instanceof WP_Post ) {
        return;
    }

    $target_slug = 'madagaskar-sirki-ankara-26-eylul-2026-1600-bileti';

    if ( $post->post_name !== $target_slug ) {
        return;
    }

    $event_url = get_permalink( $post->ID );

    /*
     * Ürünün öne çıkan görselini otomatik al.
     */
    $event_image = get_the_post_thumbnail_url( $post->ID, 'full' );

    if ( empty( $event_image ) ) {
        $event_image = 'https://i0.wp.com/madagaskarsirki.com/wp-content/uploads/2026/08/ChatGPT-Image-17-Agu-2026-09_29_49.png?ssl=1';
    }

        '@context' => 'https://schema.org',
        '@type'    => 'Event',

        'name' => 'Madagaskar Sirki Ankara',

        'description' =>
            'Hayvan kullanılmayan, uluslararası sanatçıların akrobasi, denge, komedi ve sahne performanslarıyla buluştuğu aile dostu Madagaskar Sirki gösterisi.',

        'startDate' => '2026-09-26T16:00:00+03:00',

        'eventStatus' => 'https://schema.org/EventScheduled',

        'eventAttendanceMode' =>
            'https://schema.org/OfflineEventAttendanceMode',

        'location' => array(
            '@type' => 'Place',

            'name' => 'Altınpark ANFA Fuar Alanı B Salonu',

                'addressLocality'  => 'Altındağ',
                'addressRegion'    => 'Ankara',
                'addressCountry'   => 'TR',
            ),
        ),

        'image' => array(
            $event_image,
        ),

        'offers' => array(
            '@type'         => 'Offer',
            'url'           => $event_url,
            'price'         => '250',
            'priceCurrency' => 'TRY',
            'availability'  => 'https://schema.org/InStock',
        ),

        'performer' => array(
            '@type' => 'PerformingGroup',
            'name'  => 'Madagaskar Sirki',
        ),

        'organizer' => array(
            '@type' => 'Organization',
            'name'  => 'Madagaskar Sirki Türkiye',
            'url'   => home_url( '/' ),
        ),
    );

    ?>
    <script type="application/ld+json" id="ms-ankara-event-schema-1600">

    echo wp_json_encode(
        $schema,
        JSON_UNESCAPED_UNICODE |
        JSON_UNESCAPED_SLASHES |

    );

}, 20 );
