   6. FAQ SCHEMA
   ============================================================ */

add_action(
    'wp_head',
    function () {

        if ( ! ms_sss_v3_is_page() ) {
            return;
        }

        $faqs =
            ms_sss_v3_data();

        $questions =
            array();


        foreach ( $faqs as $faq ) {

            $questions[] =
                array(

                    '@type' =>
                        'Question',

                    'name' =>
                        wp_strip_all_tags(
                            $faq['question']
                        ),

                    'acceptedAnswer' =>
                        array(

                            '@type' =>
                                'Answer',

                            'text' =>
                                wp_strip_all_tags(
                                    $faq['answer']
                                ),
                        ),
                );
        }


        $schema =
            array(

                '@context' =>
                    'https://schema.org',

                '@type' =>
                    'FAQPage',

                'mainEntity' =>
                    $questions,
            );

        ?>

<script type="application/ld+json">
<?php
echo wp_json_encode(
    $schema,
    JSON_UNESCAPED_UNICODE |
    JSON_UNESCAPED_SLASHES
);
?>
</script>

        <?php
    },
    20
);

/* ============================================================
   8. JAVASCRIPT
   ACCORDION + FILTER + SEARCH
   ============================================================ */

add_action(
    'wp_footer',
    function () {

        if ( ! ms_sss_v3_is_page() ) {
            return;
        }

        ?>

<script id="ms-sss-v3-js">

document.addEventListener(
    'DOMContentLoaded',
    function () {

        const page =
            document.getElementById(
                'ms-sss-v3'
            );

        if ( ! page ) {
            return;
        }


        const items =
            Array.from(
                page.querySelectorAll(
                    '.mss-item'
                )
            );


        const buttons =
            page.querySelectorAll(
                '.mss-question'
            );


        const filters =
            page.querySelectorAll(
                '.mss-filters button'
            );


        const input =
            document.getElementById(
                'mss-search-input'
            );


        const noResults =
            document.getElementById(
                'mss-no-results'
            );


        let activeFilter =
            'all';



        /* =====================================================
           ACCORDION
           ===================================================== */

        buttons.forEach(
            function ( button ) {

                button.addEventListener(
                    'click',
                    function () {

                        const item =
                            button.closest(
                                '.mss-item'
                            );


                        const wasOpen =
                            item.classList.contains(
                                'open'
                            );


                        items.forEach(
                            function ( other ) {

                                if (
                                    other !== item
                                ) {

                                    other.classList.remove(
                                        'open'
                                    );


                                    const otherButton =
                                        other.querySelector(
                                            '.mss-question'
                                        );


                                    if ( otherButton ) {

                                        otherButton.setAttribute(
                                            'aria-expanded',
                                            'false'
                                        );
                                    }
                                }

                            }
                        );


                        if ( wasOpen ) {

                            item.classList.remove(
                                'open'
                            );


                            button.setAttribute(
                                'aria-expanded',
                                'false'
                            );

                        } else {

                            item.classList.add(
                                'open'
                            );


                            button.setAttribute(
                                'aria-expanded',
                                'true'
                            );
                        }

                    }
                );

            }
        );



        /* =====================================================
           TÜRKÇE ARAMA NORMALİZASYONU
           ===================================================== */

        function normalizeText( value ) {

            return String(
                value || ''
            )
            .toLocaleLowerCase(
                'tr-TR'
            )
            .normalize(
                'NFD'
            )
            .replace(
                /[\u0300-\u036f]/g,
                ''
            )
            .replace(
                /ı/g,
                'i'
            );
        }



        /* =====================================================
           FİLTRE
           ===================================================== */

        function applyFilter() {

            const query =
                normalizeText(
                    input
                    ? input.value.trim()
                    : ''
                );


            let visible =
                0;


            items.forEach(
                function ( item ) {


                    const category =
                        item.dataset.category || '';


                    const text =
                        normalizeText(
                            item.dataset.search || ''
                        );


                    const categoryMatch =
                        activeFilter === 'all' ||
                        category === activeFilter;


                    const searchMatch =
                        ! query ||
                        text.includes(
                            query
                        );


                    const show =
                        categoryMatch &&
                        searchMatch;


                    item.hidden =
                        ! show;


                    if ( show ) {

                        visible++;

                    } else {

                        item.classList.remove(
                            'open'
                        );


                        const button =
                            item.querySelector(
                                '.mss-question'
                            );


                        if ( button ) {

                            button.setAttribute(
                                'aria-expanded',
                                'false'
                            );
                        }

                    }

                }
            );


            if ( noResults ) {

                noResults.hidden =
                    visible > 0;
            }

        }



        /* =====================================================
           CATEGORY FILTERS
           ===================================================== */

        filters.forEach(
            function ( filter ) {

                filter.addEventListener(
                    'click',
                    function () {

                        filters.forEach(
                            function ( item ) {

                                item.classList.remove(
                                    'active'
                                );
                            }
                        );


                        filter.classList.add(
                            'active'
                        );


                        activeFilter =
                            filter.dataset.filter || 'all';


                        applyFilter();

                    }
                );

            }
        );



        /* =====================================================
           SEARCH
           ===================================================== */

        if ( input ) {

            input.addEventListener(
                'input',
                applyFilter
            );
        }

    }
);

</script>

        <?php
    },
    9999
);
