if ( ! defined( 'ABSPATH' ) ) {
    exit;
}


/**
 * Saate bağlı liste gösteren üç sayfadan biri mi?
 */
if ( ! function_exists( 'ms_lo_liste_sayfasi_mi' ) ) {
    function ms_lo_liste_sayfasi_mi() {
        if ( function_exists( 'is_admin' ) && is_admin() ) {
            return false;
        }

        if ( function_exists( 'is_front_page' ) && is_front_page() ) {
            return true;
        }

        return function_exists( 'is_page' ) && is_page( array( 'sehirler', 'bilet-al' ) );
    }
}


if ( ! function_exists( 'ms_lo_onbellek_kapat' ) ) {
    function ms_lo_onbellek_kapat() {
        if ( ! ms_lo_liste_sayfasi_mi() ) {
            return;
        }

        if ( ! defined( 'DONOTCACHEPAGE' ) ) {
            define( 'DONOTCACHEPAGE', true );
        }

        if ( function_exists( 'nocache_headers' ) && ! headers_sent() ) {
            nocache_headers();
        }
    }
}


if ( ! function_exists( 'ms_lo_uretim_etiketi' ) ) {
    function ms_lo_uretim_etiketi() {
        if ( ! ms_lo_liste_sayfasi_mi() ) {
            return;
        }

        $an = function_exists( 'wp_date' ) ? wp_date( 'c' ) : gmdate( 'c' );

        echo '<meta name="ms-sayfa-uretim" content="' . esc_attr( $an ) . '">' . "\n";
    }
}

add_action( 'template_redirect', 'ms_lo_onbellek_kapat', 1 );
add_action( 'wp_head', 'ms_lo_uretim_etiketi', 1 );
